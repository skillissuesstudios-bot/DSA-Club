// Target Sum (Two Pointers)
function findPair(arr, target) {
let l = 0, r = arr.length - 1;
while (l < r) {
let sum = arr[l] + arr[r];
if (sum === target) return [l, r];
sum < target ? l++ : r--;
}
}
